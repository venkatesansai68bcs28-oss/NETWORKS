#include <stdio.h>

#define INF 999

int main()
{
    int n, source;
    int cost[10][10], distance[10], visited[10];
    int i, j, count, min, next;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter the cost matrix:\n");
    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &cost[i][j]);

            if (cost[i][j] == 0 && i != j)
                cost[i][j] = INF;
        }
    }

    printf("Enter source node: ");
    scanf("%d", &source);

    for (i = 0; i < n; i++)
    {
        distance[i] = cost[source][i];
        visited[i] = 0;
    }

    distance[source] = 0;
    visited[source] = 1;

    for (count = 1; count < n; count++)
    {
        min = INF;
        next = -1;

        for (i = 0; i < n; i++)
        {
            if (!visited[i] && distance[i] < min)
            {
                min = distance[i];
                next = i;
            }
        }

        if (next == -1)
            break;

        visited[next] = 1;

        for (i = 0; i < n; i++)
        {
            if (!visited[i] &&
                distance[next] + cost[next][i] < distance[i])
            {
                distance[i] = distance[next] + cost[next][i];
            }
        }
    }

    printf("\nShortest distances from node %d:\n", source);

    for (i = 0; i < n; i++)
    {
        printf("Node %d = %d\n", i, distance[i]);
    }

    return 0;
}
